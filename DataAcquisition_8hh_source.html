100 bool init_camera_reader(int32_t camera_index, ParameterStore &param_store)
101 {
102 std::unique_lock<std::mutex> acq_lock_ul{acq_lock};
103
104 // Sanity check
105 if (scanned_cameras.empty() || camera_index < 0 || camera_index >= scanned_cameras.size())
106 {
107 acq_lock_ul.unlock();
108 return false;
109 }
110
111 try
112 {
113 data_reader_ptr = std::move(
114 dv::io::camera::open(scanned_cameras[camera_index])); // Specify move semantics for unique pointer
115 }
116 catch (...)

118 std::string pop_up_err_str{"Something went wrong with the camera for reading!"};
119 param_store.add("pop_up_err_str", pop_up_err_str);

121 return false;
122 }

124 if (data_reader_ptr->isEventStreamAvailable())

223 bool get_batch_evt_data(EventData &evt_data, ParameterStore &param_store, DataWriter &data_writer,
224 float event_discard_odds)
225 {
226 std::unique_lock<std::mutex> acq_lock_ul{acq_lock};
227 // If reader is not properly initialized, return immediately
228 if (!data_reader_ptr)
229 {
230 acq_lock_ul.unlock();
231 return false;
232 }
233 bool data_read = false;
234
235 if (event_discard_odds < 0.00001)
236 {
237 std::string pop_up_err_str{"Event Discard Odds are too low!"};
238 param_store.add("pop_up_err_str", pop_up_err_str);
239 acq_lock_ul.unlock();
240 return false;
241 }
242
243 // Discard threshold
244 float threshold{1.0f / event_discard_odds};
245
246 // https://dv-processing.inivation.com/rel_1_7/reading_data.html#read-events-from-a-file
247
248 try
249 {
250 if (data_reader_ptr->isEventStreamAvailable() && data_reader_ptr->isRunning("events"))
251 {
252 if (const auto events = data_reader_ptr->getNextEventBatch(); events.has_value())
253 {
254 // In case of persistent storage
255 // https://dv-processing.inivation.com/master/event_store.html
256 dv::EventStore event_store{};
257 for (auto &evt : events.value())
258 {
259 if (randFloat() > threshold) // Random discard
260 {
261 continue; // Discard
262 }
263 EventData::EventDatum evt_datum{
264 .x = evt.x(), .y = evt.y(), .timestamp = evt.timestamp(), .polarity = evt.polarity()};
265
266 evt_data.write_evt_data(evt_datum);
267 data_read = true;
268
269 event_store.emplace_back(evt.timestamp(), evt.x(), evt.y(), evt.polarity());
270 }
271
272 // Add to queue for persistent storage in case of persistent storage
273 if (data_writer.get_writing_event_data())
274 {
275 data_writer.add_event_store(event_store);
276 }
277 }
278 }
279 }
280 catch (...)
281 {
282 std::string pop_up_err_str{"Something went wrong with reading event data!"};
283 param_store.add("pop_up_err_str", pop_up_err_str);
284 acq_lock_ul.unlock();
285 return false;
286 }
287 acq_lock_ul.unlock();
288 return data_read;
289 }

297 bool get_batch_frame_data(EventData &evt_data, ParameterStore &param_store, DataWriter &data_writer)
298 {
299 std::unique_lock<std::mutex> acq_lock_ul{acq_lock};
300 // If reader is not initialized, return immediately
301 if (!data_reader_ptr)
302 {
303 acq_lock_ul.unlock();
304 return false;
305 }
306 bool data_read = false;
307
308 // https://dv-processing.inivation.com/rel_1_7/reading_data.html#read-frames-from-a-file
309
310 try
311 {
312 if (data_reader_ptr->isFrameStreamAvailable() && data_reader_ptr->isRunning("frames"))
313 {
314 if (const auto frame_data = data_reader_ptr->getNextFrame(); frame_data.has_value())
315 {
316 cv::Mat out;
317 // Convert frame data from BGR to RGB
318 cv::cvtColor(frame_data->image, out, cv::COLOR_BGR2RGB);
319
320 // CLone to ensure tightly packed frame bytes
321 EventData::FrameDatum frame_datum{.frameData = out.clone(), .timestamp = frame_data->timestamp};
322
323 evt_data.write_frame_data(frame_datum);
324 data_read = true;
325
326 // If saving stream, add to queue to write
327 if (data_writer.get_writing_frame_data())
328 {
329 dv::Frame frame_datum(frame_data->timestamp, frame_data->image);
330 data_writer.add_frame_data(frame_datum);
331 }
332 }
333 }
334 }
335 catch (...)
336 {
337 std::string pop_up_err_str{"Something went wrong with reading frame data!"};
338 param_store.add("pop_up_err_str", pop_up_err_str);
339 acq_lock_ul.unlock();
340 return false;
341 }
342 acq_lock_ul.unlock();
343 return data_read;
