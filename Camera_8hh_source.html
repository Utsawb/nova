238 void updateOrbitPosition()
239 {
240 // Convert to spherical coordinates
241 float pitch_rad = glm::radians(m_pitch);
242 float yaw_rad = glm::radians(m_yaw);
243
244 // Calculate position in spherical coordinates
245 // Using standard spherical coordinates with yaw=0 pointing along +X
246 float x = m_orbit_center.x + m_orbit_distance * cos(pitch_rad) * cos(yaw_rad);
247 float y = m_orbit_center.y + m_orbit_distance * sin(pitch_rad);
248 float z = m_orbit_center.z + m_orbit_distance * cos(pitch_rad) * sin(yaw_rad);
249
250 m_position = glm::vec3(x, y, z);
251
252 // Update front vector to look at the orbit center
253 m_front = glm::normalize(m_orbit_center - m_position);
254
255 // Recalculate right and up vectors
256 m_right = glm::normalize(glm::cross(m_front, m_world_up));
257 m_up = glm::normalize(glm::cross(m_right, m_front));
258 }
