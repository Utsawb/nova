188 // if an event was handled, return true, otherwise return false
189 bool event_handler(SDL_Event *event)
190 {
191 if (render_targets["SpinningCubeColor"].is_focused == true)
192 {
193 switch (event->type)
194 {
195 case SDL_EVENT_MOUSE_BUTTON_DOWN:
196 if (event->button.button == SDL_BUTTON_LEFT)
197 {
198 is_mouse_dragging = true;
199 last_mouse_x = event->button.x;
200 last_mouse_y = event->button.y;
201
202 // Hide cursor and enable relative mouse mode for unlimited movement
203 SDL_HideCursor();
204 SDL_SetWindowRelativeMouseMode(window, true);
205 cursor_captured = true;
206 }
207 break;
208
209 case SDL_EVENT_MOUSE_BUTTON_UP:
210 if (event->button.button == SDL_BUTTON_LEFT)
211 {
212 is_mouse_dragging = false;
213
214 // Restore cursor and disable relative mouse mode
215 if (cursor_captured)
216 {
217 SDL_SetWindowRelativeMouseMode(window, false);
218 SDL_ShowCursor();
219 cursor_captured = false;
220 }
221 }
222 break;
223
224 case SDL_EVENT_MOUSE_MOTION:
225 if (is_mouse_dragging && cursor_captured)
226 {
227 // When relative mouse mode is enabled, motion.x and motion.y contain
228 // the relative movement from the last motion event
229 float x_offset = event->motion.xrel;
230 float y_offset = event->motion.yrel;
231
232 // Update camera rotation based on mouse movement
233 camera.processMouseMovement(x_offset, y_offset);
234 }
235 break;
236
237 case SDL_EVENT_WINDOW_FOCUS_LOST:
238 // If cursor is captured and window loses focus, restore it
239 if (cursor_captured)
240 {
241 SDL_SetWindowRelativeMouseMode(window, false);
242 SDL_ShowCursor();
243 cursor_captured = false;
244 is_mouse_dragging = false;
245 }
246 break;
247
248 default:
249 break;
250 }
251 render_targets["SpinningCubeColor"].is_focused = false;
252 return true;
253 }
254 return false;

257 void cpu_update()
258 {
259 // Update rotation time (assuming 60 FPS, adjust as needed)
260 // rotation_time += 0.016f; // ~60 FPS
261
262 // Create model matrix with rotation
263 glm::mat4 model = glm::mat4(1.0f);
264 model = glm::rotate(model, rotation_time, glm::vec3(1.0f, 1.0f, 0.0f)); // Rotate around X+Y axis
265
266 // Get view and projection matrices from camera
267 glm::mat4 view = camera.getViewMatrix();
268 glm::mat4 projection = camera.getProjectionMatrix();
269
270 // Combine matrices: MVP = Projection * View * Model
271 mvp = projection * view * model;
272 }
