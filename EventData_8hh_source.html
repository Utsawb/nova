174 void ensure_capacity(std::size_t min_capacity)
175 {
176 if (min_capacity <= capacity_)
177 {
178 return;
179 }
180
181 std::size_t new_capacity = capacity_ == 0 ? kInitialCapacity : capacity_;
182 while (new_capacity < min_capacity)
183 {
184 new_capacity *= 2;
185 }
186
187 remap(new_capacity);
188 }
189
190 void remap(std::size_t new_capacity)
191 {
192 const std::size_t bytes = new_capacity * sizeof(value_type);
193
194 mapped_file_.close();
195
196 if (bytes == 0)
197 {
198 capacity_ = 0;
199 return;
200 }
201
202 {
203 std::fstream file(file_path_, std::ios::binary | std::ios::in | std::ios::out);
204 if (!file)
205 {
206 std::ofstream create(file_path_, std::ios::binary | std::ios::out | std::ios::trunc);
207 create.close();
208 file.open(file_path_, std::ios::binary | std::ios::in | std::ios::out);
209 }
210
211 file.seekp(static_cast<std::streamoff>(bytes) - 1, std::ios::beg);
212 char zero = 0;
213 file.write(&zero, 1);
214 }
215
216 boost::iostreams::mapped_file_params params;
217 params.path = file_path_.string();
218 params.mode = std::ios_base::in | std::ios_base::out;
219 params.length = bytes;
220 mapped_file_.open(params);
221 if (!mapped_file_.is_open())
222 {
223 throw std::runtime_error("Failed to open mapped file for EventData buffer.");
224 }
225 capacity_ = new_capacity;
226 }
227
228 void erase_front(std::size_t count)
229 {
230 if (count == 0 || count > size_)
231 {
232 return;
233 }
234
235 value_type *base_ptr = ptr();
236 std::memmove(base_ptr, base_ptr + count, (size_ - count) * sizeof(value_type));
237 size_ -= count;
238 }

402 void write_evt_data(EventDatum raw_evt_data)
403 {
404 std::unique_lock<std::recursive_mutex> evt_lock_ul{evt_lock};
405
406 constexpr size_t MAX_EVENT_BACKING_SIZE{
407 100 * ((static_cast<size_t>(1) << 30) / (1 << 4))}; // Set 100 GB approximately
408
409 // If this condition is met, then we have unordered data, assume camera reset, clear all previous data
410 // Or maximum number of event data has been reached
411 if ((raw_evt_data.timestamp < evt_data_latest_timestamp) || (evt_data_vector_relative.size() > MAX_EVENT_BACKING_SIZE))
412 {
413 // Reset assumed, timestamps are all back to zero, clear data
414 evt_data_earliest_timestamp = -1;
415 evt_data_latest_timestamp = -1;
416 frame_data_latest_timestamp = -1;
417 evt_data_vector_relative.clear();
418 frame_data_vector_relative.clear();
419 }
420
421 // update earliest timestamp
422 if (evt_data_vector_relative.empty())
423 {
424 evt_data_earliest_timestamp = raw_evt_data.timestamp;
425 }
426
427 float x{static_cast<float>(raw_evt_data.x)};
428 float y{static_cast<float>(raw_evt_data.y)};
429 float timestamp_relative{static_cast<float>(raw_evt_data.timestamp - evt_data_earliest_timestamp)};
430 float polarity{static_cast<float>(raw_evt_data.polarity)};
431 evt_data_vector_relative.push_back(glm::vec4{x, y, timestamp_relative, polarity});
432
433 evt_data_latest_timestamp = raw_evt_data.timestamp;
434
435 evt_lock_ul.unlock();
436 }

446 void write_frame_data(FrameDatum raw_frame_data)
447 {
448 std::unique_lock<std::recursive_mutex> evt_lock_ul{evt_lock};
449
450 constexpr size_t MAX_FRAME_SIZE{static_cast<size_t>(1) << 20}; // Set max to 1 million frames
451
452 // If this condition is met, unordered frame assumes camera reset,
453 // or maximum number of frames met.
454 if ((raw_frame_data.timestamp < frame_data_latest_timestamp) || (frame_data_vector_relative.size() > MAX_FRAME_SIZE))
455 {
456 // Reset assumed, timestamps are all back to zero, clear data
457 evt_data_earliest_timestamp = -1;
458 evt_data_latest_timestamp = -1;
459 frame_data_latest_timestamp = -1;
460 evt_data_vector_relative.clear();
461 frame_data_vector_relative.clear();
462 }
463
464 // update earliest timestamp
465 if (evt_data_vector_relative.empty()) // Need to normalize timestamps relative to event data, ignore until event data is in
466 {
467 return;
468 }
469
470 float timestamp_relative{static_cast<float>(raw_frame_data.timestamp - evt_data_earliest_timestamp)};
471
472 frame_data_vector_relative.push_back(std::make_pair(raw_frame_data.frameData, timestamp_relative));
473
474 frame_data_latest_timestamp = raw_frame_data.timestamp;
475
476 evt_lock_ul.unlock();
477 }

532 int64_t get_event_index_from_relative_timestamp(float timestamp)
533 {
534 std::unique_lock<std::recursive_mutex> evt_lock_ul{evt_lock};
535
536 if (evt_data_vector_relative.empty())
537 {
538 evt_lock_ul.unlock();
539 return -1; // Vector is empty, return -1
540 }
541
542 glm::vec4 timestampVec4(0.0f, 0.0f, timestamp, 0.0f);
543
544 // evt_data_vector_relative should be sorted by the way EventData class is setup.
545 auto lb = std::lower_bound(evt_data_vector_relative.begin(), evt_data_vector_relative.end(), timestampVec4,
546 event_less_vec4_t);
547 if (lb == evt_data_vector_relative.end())
548 {
549 return -1; // Not found
550 }
551 int64_t ret_index{static_cast<int64_t>(std::distance(evt_data_vector_relative.begin(), lb))};
552
553 evt_lock_ul.unlock();
554 return ret_index;
