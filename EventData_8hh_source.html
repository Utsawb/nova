277 void ensure_capacity(std::size_t min_capacity)
278 {
279 if (min_capacity <= capacity_)
280 {
281 return;
282 }
283
284 std::size_t new_capacity = capacity_ == 0 ? kInitialCapacity : capacity_;
285 while (new_capacity < min_capacity)
286 {
287 new_capacity *= 2;
288 }
289
290 remap(new_capacity);
291 }
292
298 void remap(std::size_t new_capacity)
299 {
300 const std::size_t bytes = new_capacity * sizeof(value_type);
301
302 mapped_file_.close();
303
304 if (bytes == 0)
305 {
306 capacity_ = 0;
307 return;
308 }
309
310 {
311 std::fstream file(file_path_, std::ios::binary | std::ios::in | std::ios::out);
312 if (!file)
313 {
314 std::ofstream create(file_path_, std::ios::binary | std::ios::out | std::ios::trunc);
315 create.close();
316 file.open(file_path_, std::ios::binary | std::ios::in | std::ios::out);
317 }
318
319 file.seekp(static_cast<std::streamoff>(bytes) - 1, std::ios::beg);
320 char zero = 0;
321 file.write(&zero, 1);
322 }
323
324 boost::iostreams::mapped_file_params params;
325 params.path = file_path_.string();
326 params.mode = std::ios_base::in | std::ios_base::out;
327 params.length = bytes;
328 mapped_file_.open(params);
329 if (!mapped_file_.is_open())
330 {
331 throw std::runtime_error("Failed to open mapped file for EventData buffer.");
332 }
333 capacity_ = new_capacity;
334 }

523 void write_evt_data(EventDatum raw_evt_data)
524 {
525 std::unique_lock<std::recursive_mutex> evt_lock_ul{evt_lock};
526
527 // 100 GB / sizeof(glm::vec4) = max number of elements to reach 100 GB
528 constexpr size_t MAX_EVENT_BACKING_SIZE{
529 (static_cast<size_t>(100) * (static_cast<size_t>(1) << 30)) /
530 sizeof(glm::vec4)}; // Set 100 GB approximately as maximum of backing store
531
532 // If this condition is met, then we have unordered data, assume camera reset, clear all previous data
533 // Or maximum number of event data has been reached
534 if ((raw_evt_data.timestamp < evt_data_latest_timestamp) ||
535 (evt_data_vector_relative.size() > MAX_EVENT_BACKING_SIZE))
536 {
537 // Reset assumed, timestamps are all back to zero, clear data
538 evt_data_earliest_timestamp = -1;
539 evt_data_latest_timestamp = -1;
540 frame_data_latest_timestamp = -1;
541 evt_data_vector_relative.clear();
542 frame_data_vector_relative.clear();
543 }
544
545 // update earliest timestamp
546 if (evt_data_vector_relative.empty())
547 {
548 evt_data_earliest_timestamp = raw_evt_data.timestamp;
549 }
550
551 float x{static_cast<float>(raw_evt_data.x)};
552 float y{static_cast<float>(raw_evt_data.y)};
553 float timestamp_relative{static_cast<float>(raw_evt_data.timestamp - evt_data_earliest_timestamp)};
554 float polarity{static_cast<float>(raw_evt_data.polarity)};
555 evt_data_vector_relative.push_back(glm::vec4{x, y, timestamp_relative, polarity});
556
557 evt_data_latest_timestamp = raw_evt_data.timestamp;
558
559 evt_lock_ul.unlock();
560 }

570 void write_frame_data(FrameDatum raw_frame_data)
571 {
572 std::unique_lock<std::recursive_mutex> evt_lock_ul{evt_lock};
573
574 constexpr size_t MAX_FRAME_SIZE{static_cast<size_t>(1) << 20}; // Set max to 1 million frames
575
576 // If this condition is met, unordered frame assumes camera reset,
577 // or maximum number of frames met.
578 if ((raw_frame_data.timestamp < frame_data_latest_timestamp) ||
579 (frame_data_vector_relative.size() > MAX_FRAME_SIZE))
580 {
581 // Reset assumed, timestamps are all back to zero, clear data
582 evt_data_earliest_timestamp = -1;
583 evt_data_latest_timestamp = -1;
584 frame_data_latest_timestamp = -1;
585 evt_data_vector_relative.clear();
586 frame_data_vector_relative.clear();
587 }
588
589 // update earliest timestamp
590 if (evt_data_vector_relative
591 .empty()) // Need to normalize timestamps relative to event data, ignore until event data is in
592 {
593 return;
594 }
595
596 float timestamp_relative{static_cast<float>(raw_frame_data.timestamp - evt_data_earliest_timestamp)};
597
598 frame_data_vector_relative.push_back(std::make_pair(raw_frame_data.frameData, timestamp_relative));
599
600 frame_data_latest_timestamp = raw_frame_data.timestamp;
601
602 evt_lock_ul.unlock();
603 }

654 int64_t get_event_index_from_relative_timestamp(float timestamp)
655 {
656 std::unique_lock<std::recursive_mutex> evt_lock_ul{evt_lock};
657
658 if (evt_data_vector_relative.empty())
659 {
660 evt_lock_ul.unlock();
661 return -1; // Vector is empty, return -1
662 }
663
664 glm::vec4 timestampVec4(0.0f, 0.0f, timestamp, 0.0f);
665
666 // evt_data_vector_relative should be sorted by the way EventData class is setup.
667 auto lb = std::lower_bound(evt_data_vector_relative.begin(), evt_data_vector_relative.end(), timestampVec4,
668 event_less_vec4_t);
669 if (lb == evt_data_vector_relative.end())
670 {
671 return -1; // Not found
672 }
673 int64_t ret_index{static_cast<int64_t>(std::distance(evt_data_vector_relative.begin(), lb))};
674
